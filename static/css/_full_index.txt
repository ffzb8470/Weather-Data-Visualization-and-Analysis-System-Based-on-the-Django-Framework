<div class="city-grid" id="cityGrid"></div>

<script>
const CITIES = {{ cities|safe }};
const COLORS = {{ colors|safe }};
const DAILY_DATA = {{ daily_data|safe }};
const LATEST_DATA = {{ latest_json|safe }};

function getQuality(aqi) {
    if (aqi == null) return {text:'无数据', cls:'#999'};
    if (aqi <= 50) return {text:'优', cls:'#00e400'};
    if (aqi <= 100) return {text:'良', cls:'#ffff00', tc:'#333'};
    if (aqi <= 150) return {text:'轻度污染', cls:'#ff7e00'};
    if (aqi <= 200) return {text:'中度污染', cls:'#ff0000'};
    if (aqi <= 300) return {text:'重度污染', cls:'#99004c'};
    return {text:'严重污染', cls:'#7e0023'};
}
function darken(hex, amt) {
    let c = parseInt(hex.replace('#',''), 16);
    let r = Math.max((c>>16)-amt, 0);
    let g = Math.max(((c>>8)&0xff)-amt, 0);
    let b = Math.max((c&0xff)-amt, 0);
    return `rgb(${r},${g},${b})`;
}

function renderNormalCard(city, rec, color) {
    const aqi = rec.aqi != null ? rec.aqi : '--';
    const q = getQuality(aqi);
    const dark = darken(color, 40);
    const fields = [
        {f:'pm25',l:'PM2.5 (μg/m³)'},{f:'pm10',l:'PM10 (μg/m³)'},
    ];
    let pollHtml = fields.map(f => `
        <div class="p-item">
            <span class="p-label">${f.l}</span>
            <span class="p-value">${rec[f.f] != null ? rec[f.f] : '--'}</span>
        </div>
    `).join('');

    const card = document.createElement('div');
    card.className = 'city-card card-enter';
    card.dataset.city = city;
    card.innerHTML = `
        <div class="card-top" style="background:linear-gradient(135deg,${color},${dark})">
            <div class="city-name">${city} <span class="date-badge">${rec.date || ''}</span></div>
        </div>
        <div class="card-body">
            <div class="aqi-ring" style="background:radial-gradient(circle at 30% 30%,${color}88,${dark})">
                <span class="aqi-num">${aqi}</span>
                <span class="aqi-unit">AQI</span>
            </div>
            <span class="quality-tag" style="background:${q.cls};${q.tc ? 'color:'+q.tc : ''}">${q.text}</span>
            <div class="pollutants">${pollHtml}</div>
        </div>
    `;
    card.addEventListener('click', () => expandCity(city));
    return card;
}

function renderMinimizedCard(city, rec, color) {
    const aqi = rec.aqi != null ? rec.aqi : '--';
    const dark = darken(color, 40);
    const card = document.createElement('div');
    card.className = 'city-card minimized card-enter';
    card.dataset.city = city;
    card.innerHTML = `
        <div class="aqi-ring" style="background:radial-gradient(circle at 30% 30%,${color}88,${dark})">
            <span class="aqi-num">${aqi}</span>
        </div>
        <div class="city-name">${city}</div>
    `;
    card.addEventListener('click', () => expandCity(city));
    return card;
}

function renderExpandedCard(city, rec, color) {
    const aqi = rec.aqi != null ? rec.aqi : '--';
    const q = getQuality(aqi);
    const dark = darken(color, 40);
    const fields = [
        {f:'pm25',l:'PM2.5 (μg/m³)'},{f:'pm10',l:'PM10 (μg/m³)'},
        {f:'no2',l:'NO₂ (μg/m³)'},{f:'so2',l:'SO₂ (μg/m³)'},
        {f:'co',l:'CO (mg/m³)'},{f:'o3',l:'O₃ (μg/m³)'}
    ];
    let pollHtml = fields.map(f => `
        <div class="p-item">
            <span class="p-label">${f.l}</span>
            <span class="p-value">${rec[f.f] != null ? rec[f.f] : '--'}</span>
        </div>
    `).join('');

    const card = document.createElement('div');
    card.className = 'city-card expanded card-enter';
    card.dataset.city = city;
    card.innerHTML = `
        <button class="close-btn">✕</button>
        <div class="card-top" style="background:linear-gradient(135deg,${color},${dark})">
            <div class="city-name">${city} <span class="date-badge">${rec.date || ''}</span></div>
        </div>
        <div class="card-body">
            <div class="left-col">
                <div class="aqi-ring" style="background:radial-gradient(circle at 30% 30%,${color}88,${dark})">
                    <span class="aqi-num">${aqi}</span>
                    <span class="aqi-unit">AQI</span>
                </div>
                <span class="quality-tag" style="background:${q.cls};${q.tc ? 'color:'+q.tc : ''}">${q.text}</span>
                <div class="pollutants">${pollHtml}</div>
                <div style="margin-top:10px;display:flex;gap:8px;">
                    <button class="tab-btn active" data-tab="daily-${city}">近30天</button>
                    <button class="tab-btn" data-tab="pie-${city}">污染物占比</button>
                </div>
            </div>
            <div style="position:relative;">
                <div class="chart-container-sm" id="daily-${city}"></div>
                <div class="chart-container-sm" id="pie-${city}" style="display:none;"></div>
            </div>
        </div>
    `;

    card.querySelector('.close-btn').addEventListener('click', (e) => {
        e.stopPropagation();
        resetToNormal();
    });

    card.querySelectorAll('.tab-btn').forEach(btn => {
        btn.addEventListener('click', (e) => {
            e.stopPropagation();
            const tab = btn.dataset.tab;
            card.querySelectorAll('.tab-btn').forEach(b => b.classList.remove('active'));
            btn.classList.add('active');
            card.querySelectorAll('.chart-container-sm').forEach(el => el.style.display = 'none');
            document.getElementById(tab).style.display = '';
            if (tab.startsWith('daily-')) drawDailyChart(city);
            else if (tab.startsWith('pie-')) drawPieChart(city);
        });
    });

    return card;
}

function expandCity(city) {
    const grid = document.getElementById('cityGrid');

    // === FLIP: 记录旧卡片的位置 ===
    const oldCard = document.querySelector(`.city-card[data-city="${city}"]`);
    const oldRect = oldCard ? oldCard.getBoundingClientRect() : null;

    grid.classList.add('expanded');

    // 清除旧卡片
    grid.innerHTML = '';

    // 被选中的卡片置顶
    const rec = LATEST_DATA[city] || {};
    const color = COLORS[city] || '#37A2DA';
    const expandedCard = renderExpandedCard(city, rec, color);
    grid.appendChild(expandedCard);

    // 其他卡片缩小排在后面
    CITIES.forEach((c, i) => {
        if (c === city) return;
        const rec2 = LATEST_DATA[c] || {};
        const color2 = COLORS[c] || '#37A2DA';
        const card = renderMinimizedCard(c, rec2, color2);
        card.style.opacity = '0';
        card.style.transform = 'translateY(24px)';
        card.style.transition = 'none';
        grid.appendChild(card);
    });

    // === FLIP: 计算位移并播放动画 ===
    requestAnimationFrame(() => {
        const newRect = expandedCard.getBoundingClientRect();

        if (oldRect) {
            const dx = oldRect.left - newRect.left;
            const dy = oldRect.top - newRect.top;
            const sx = oldRect.width / newRect.width;
            const sy = oldRect.height / newRect.height;

            // 从旧位置开始
            expandedCard.style.transformOrigin = 'top left';
            expandedCard.style.transform = `translate(${dx}px, ${dy}px) scale(${sx}, ${sy})`;
            expandedCard.style.transition = 'none';
            expandedCard.style.opacity = '1';
        }

        // 强制重排后播放入场动画
        requestAnimationFrame(() => {
            if (oldRect) {
                expandedCard.style.transition = 'all 0.5s cubic-bezier(0.34, 1.56, 0.64, 1)';
                expandedCard.style.transform = '';
                expandedCard.style.opacity = '1';
            }

            // 缩小卡片依次滑入
            CITIES.forEach((c, i) => {
                if (c === city) return;
                const card = grid.querySelector(`.city-card[data-city="${c}"]`);
                if (!card) return;
                card.style.transition = `all 0.4s cubic-bezier(0.34, 1.56, 0.64, 1) ${0.1 + i * 0.05}s`;
                card.style.opacity = '1';
                card.style.transform = '';
            });

            // 绘制图表
            setTimeout(() => drawDailyChart(city), 50);
        });
    });

    // 平滑滚动到顶部
    window.scrollTo({ top: 0, behavior: 'smooth' });
}

function resetToNormal() {
    const grid = document.getElementById('cityGrid');
    grid.classList.remove('expanded');
    grid.innerHTML = '';
    initDashboard();
}

function initDashboard() {
    const grid = document.getElementById('cityGrid');
    grid.innerHTML = '';
    CITIES.forEach(city => {
        const rec = LATEST_DATA[city] || {};
        const color = COLORS[city] || '#37A2DA';
        const card = renderNormalCard(city, rec, color);
        grid.appendChild(card);
    });
}

function drawDailyChart(city) {
    const container = document.getElementById('daily-' + city);
    if (!container) return;

    let chart = echarts.getInstanceByDom(container);
    if (!chart) chart = echarts.init(container);

    const daily = DAILY_DATA[city] || {};
    const dates = Object.keys(daily);
    const aqiData = dates.map(d => daily[d].aqi);

    const C2 = getChartColors();
    chart.setOption({
        tooltip: { trigger: 'axis' },
        grid: { left: 35, right: 8, top: 10, bottom: 20 },
        xAxis: {
            type: 'category',
            data: dates,
            axisLabel: { fontSize: 9, rotate: 30, interval: Math.max(1, Math.floor(dates.length / 15)), color: C2.subText },
            axisLine: { lineStyle: { color: C2.splitLine } }
        },
        yAxis: {
            type: 'value',
            splitLine: { lineStyle: { type: 'dashed', color: C2.splitLine } }
        },
        series: [{
            type: 'line',
            smooth: true,
            data: aqiData,
            lineStyle: { width: 2, color: COLORS[city] || '#37A2DA' },
            areaStyle: {
                color: {
                    type: 'linear', x: 0, y: 0, x2: 0, y2: 1,
                    colorStops: [
                        { offset: 0, color: (COLORS[city] || '#37A2DA') + '88' },
                        { offset: 1, color: (COLORS[city] || '#37A2DA') + '11' }
                    ]
                }
            },
            symbol: 'none',
        }]
    });
    chart.resize();
}

function drawPieChart(city) {
    const container = document.getElementById('pie-' + city);
    if (!container) return;

    let chart = echarts.getInstanceByDom(container);
    if (!chart) chart = echarts.init(container);

    const rec = LATEST_DATA[city] || {};
    const fields = [
        { name: 'PM2.5', value: rec.pm25 },
        { name: 'PM10', value: rec.pm10 },
        { name: 'NO₂', value: rec.no2 },
        { name: 'SO₂', value: rec.so2 },
        { name: 'CO', value: rec.co != null ? rec.co * 1000 : null },
        { name: 'O₃', value: rec.o3 }
    ];
    const pieData = fields.filter(f => f.value != null && f.value > 0);
    if (pieData.length === 0) return;

    const C2 = getChartColors();
    chart.setOption({
        tooltip: {
            trigger: 'item',
            formatter: '{b}: {c} μg/m³ ({d}%)'
        },
        series: [{
            type: 'pie',
            radius: ['30%', '70%'],
            center: ['50%', '50%'],
            data: pieData.map(f => ({
                name: f.name,
                value: f.value
            })),
            label: {
                show: true,
                formatter: '{b}\n{d}%',
                fontSize: 10,
                color: C2.text
            },
            emphasis: {
                itemStyle: {
                    shadowBlur: 10,
                    shadowOffsetX: 0,
                    shadowColor: C2.text
                }
            }
        }]
    });
    chart.resize();
}

initDashboard();
</script>
